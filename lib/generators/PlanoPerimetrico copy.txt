import { jsPDF } from 'jspdf';
import { GenerarPlanosRequest, PlanoConfig } from '@/types/planos';
import { CADDrawing } from '@/lib/geometry/cadDrawing';
import { utmToPaper, utmToPaperRelative, metrosAPapel } from '@/lib/geometry/scaleUtils';
import { calculateDistance, getBoundingBox, calculateCentroid, utmToLatLng } from '@/lib/geometry/utmUtils';
import { MapService } from '@/lib/services/MapService';

/**
 * Generador de Plano Perimétrico Profesional (Versión Mejorada)
 * Soporta renderizado híbrido con prioridad:
 * 1. Vectores (Contexto Urbano)
 * 2. Imagen Base64 (Snapshot Leaflet)
 * 3. Mapa Satelital (Google/Server)
 */
export class PlanoPerimetricoGenerator {
  private request: GenerarPlanosRequest;

  constructor(
    request: GenerarPlanosRequest,
    private config: PlanoConfig
  ) {
    this.request = request;
  }

  async generate(pdf: jsPDF): Promise<void> {
    const { vertices, lote, dimensiones, colindancias, contexto, imagenContexto, propietario } = this.request as any;
    const cad = new CADDrawing(pdf);

    const pageWidth = pdf.internal.pageSize.width;
    const pageHeight = pdf.internal.pageSize.height;

    // ========== 1. DEFINICIÓN DE ÁREAS (LAYOUT) ==========
    const rightColumnWidth = 135;
    const margin = 10;
    const gap = 5;

    // Ancho disponible para el dibujo principal
    const drawingWidthRaw = pageWidth - rightColumnWidth - (margin * 2) - gap;
    
    const drawingArea = {
      x: margin,
      y: margin,
      width: drawingWidthRaw,
      height: pageHeight - (margin * 2)
    };

    // Áreas de la Columna Derecha
    const colX = pageWidth - rightColumnWidth - margin;
    const ubicacionHeight = 80;
    
    // --- CAMBIOS AQUÍ ---
    const membreteHeight = 60; // Reducimos altura (antes 70) porque el título va afuera
    const titleBarHeight = 10; // Nueva altura para la barra de título
    
    // Recalculamos el espacio del cuadro técnico
    const cuadroTecnicoHeight = pageHeight - (margin * 2) - membreteHeight - titleBarHeight - ubicacionHeight - (gap * 3); // gap extra
    
    const ubicacionArea = { x: colX, y: margin, width: rightColumnWidth, height: ubicacionHeight };
    const technicalTableArea = { x: colX, y: margin + ubicacionHeight + gap, width: rightColumnWidth, height: cuadroTecnicoHeight };
    
    // El membrete va abajo del todo
    const membreteArea = { x: colX, y: pageHeight - membreteHeight - margin, width: rightColumnWidth, height: membreteHeight };
    
    // La barra de título va JUSTO ENCIMA del membrete
    const titleBarArea = { x: colX, y: membreteArea.y - titleBarHeight, width: rightColumnWidth, height: titleBarHeight };
    // --------------------

    // ========== 2. MARCO GLOBAL ==========
    this.drawProfessionalBorder(pdf, pageWidth, pageHeight);

    // ========== 3. CÁLCULO DE ESCALA Y GEOMETRÍA ==========
    // Calculamos escala basada en el lote objetivo (+ margen 20mm) para que sea el protagonista
    const { escala, escalaTexto } = this.calculateScaleForViewport(vertices, drawingArea.width, drawingArea.height, 20);
    
    // Centro del área de dibujo en papel
    const centerX = drawingArea.x + drawingArea.width / 2;
    const centerY = drawingArea.y + drawingArea.height / 2;

    // ========== 4. RENDERIZADO DEL CONTEXTO (FONDO) ==========
    // Lógica de cascada de prioridades: Vectores > Imagen > Mapa
    let fondoDibujado = false;
    let esFondoOscuro = false; // Para decidir si usamos relleno blanco en etiquetas

    // PRIORIDAD 1: VECTORES (Coordenadas de vecinos)
    // PRIORIDAD 1: VECTORES (Coordenadas de vecinos)
    // Verificamos lotesVecinos (Schema oficial) O elementos (Schema legacy/flexible)
    const vecinos = contexto?.lotesVecinos || (contexto as any)?.elementos;
    
    if (vecinos && vecinos.length > 0) {
      // console.log('Modo: Contexto Vectorial', vecinos.length);
      this.renderVectorialContext(pdf, cad, vecinos, escala, centerX, centerY);
      fondoDibujado = true;
    }
    // PRIORIDAD 2: IMAGEN (Captura de pantalla Leaflet enviada por Frontend)
    else if (imagenContexto && imagenContexto.data) {
      // console.log('Modo: Captura de Pantalla');
      this.renderImageContext(pdf, imagenContexto.data, drawingArea);
      fondoDibujado = true;
      esFondoOscuro = true; // Asumimos que la imagen satelital es oscura/compleja
    }
    // PRIORIDAD 3: MAPA SERVER (Google Static Maps fallback)
    else {
      // console.log('Modo: Mapa Servidor (Fallback)');
      await this.renderMapServiceContext(pdf, drawingArea, vertices);
      fondoDibujado = true;
      esFondoOscuro = true;
    }

    // ========== 5. RENDERIZADO TÉCNICO (CAPA SUPERIOR) ==========
    
    // A. Grilla UTM (Siempre encima del fondo)
    this.drawRealUTMGrid(pdf, drawingArea, vertices, escala, centerX, centerY);

    // B. Lote Principal (El Protagonista)
    // Convertimos vértices UTM a coordenadas del papel
    const paperPoints = utmToPaper(vertices, escala, centerX, centerY);
    
    cad.drawPolygon(paperPoints, {
      lineWidth: 0.7,         // Línea gruesa profesional
      strokeColor: '#000000', // Negro puro
      // Si hay vecinos vectoriales, rellenamos suave. Si hay mapa de fondo, transparente.
      fillColor: ((contexto as any)?.elementos?.length ?? 0) > 0 ? '#EFEFEF' : undefined
    });

    // C. Datos Centrales (Área/Perímetro)
    this.drawPolygonCentralData(pdf, paperPoints, lote, dimensiones);

    // D. Datos Topográficos (Ángulos, Cotas, Vértices)
    const datosTopograficos = this.calculateTopographicData(vertices);
    this.drawVerticesAndAngles(pdf, paperPoints, datosTopograficos.angulos, cad);
    this.drawDimensions(pdf, paperPoints, vertices, esFondoOscuro);

    // E. Etiquetas de Colindancia (Calles, etc.)
    if (this.config.incluirColindantesEnPlano && colindancias.length > 0) {
      this.drawBoundaryLabels(pdf, paperPoints, colindancias, esFondoOscuro);
    }
    
    // F. Norte y Título
    this.drawNorthCatastro(pdf, drawingArea.x + drawingArea.width - 25, drawingArea.y + 35, 20);
    this.drawTitle(pdf, drawingArea, 'PLANO PERIMÉTRICO');

    // ========== 6. COLUMNA DERECHA ==========
    this.drawLocationPlan(pdf, ubicacionArea, vertices);
    this.drawTechnicalTable(pdf, technicalTableArea, vertices, datosTopograficos, escalaTexto);
    
    // NUEVO: Dibujamos la barra de título separada
    this.drawHeaderTitleBar(pdf, titleBarArea);
    
    this.drawProfessionalMembrete(pdf, membreteArea, lote, dimensiones, escalaTexto, propietario);
  }

  // ===========================================================================
  // MÉTODOS DE RENDERIZADO DE CONTEXTO
  // ===========================================================================

  /** PRIORIDAD 1: Dibuja polígonos vecinos y sus números */
  private renderVectorialContext(pdf: jsPDF, cad: CADDrawing, elementos: any[], escala: number, cx: number, cy: number) {
    // Necesitamos el centroide del lote PRINCIPAL para usarlo como referencia relativa
    // Esto asegura que los vecinos se dibujen en la posición correcta respecto al lote central
    const { vertices } = this.request as any;
    const centroUtm = calculateCentroid(vertices);

    elementos.forEach((el) => {
      // Solo dibujamos si es tipo LOTE o AREA_VERDE
      if (el.vertices && el.vertices.length > 2) {
        // Usamos transformación relativa al centroide del lote principal
        const pts = utmToPaperRelative(el.vertices, centroUtm, escala, cx, cy);
        
        // Estilo según tipo
        const isVerde = el.tipo === 'AREA_VERDE';
        cad.drawPolygon(pts, {
          strokeColor: '#999999', // Gris medio
          lineWidth: 0.15,
          fillColor: isVerde ? '#F0FFF0' : '#FAFAFA' // Verde muy suave o gris casi blanco
        });

        // Etiqueta del lote (Ej: "13")
        if (el.texto) {
          const centro = this.calculateVisualCenter(pts);
          // Validar coordenadas antes de dibujar texto
          if (!isNaN(centro.x) && !isNaN(centro.y)) {
            pdf.setFont('helvetica', 'bold');
            pdf.setFontSize(isVerde ? 6 : 8);
            pdf.setTextColor(100, 100, 100);
            pdf.text(el.texto, centro.x, centro.y, { align: 'center', baseline: 'middle' });
          }
        }
      }
    });
  }

  /** PRIORIDAD 2: Dibuja imagen base64 con capa de blanqueo */
  private renderImageContext(pdf: jsPDF, base64: string, area: any) {
    try {
      pdf.addImage(base64, 'PNG', area.x, area.y, area.width, area.height);
      // Capa blanca semitransparente para que resalte el dibujo técnico
      pdf.saveGraphicsState();
      pdf.setGState(new (pdf as any).GState({ opacity: 0.25 }));
      pdf.setFillColor(255, 255, 255);
      pdf.rect(area.x, area.y, area.width, area.height, 'F');
      pdf.restoreGraphicsState();
    } catch (e) {
      console.error('Error renderizando imagen contexto:', e);
    }
  }

  /** PRIORIDAD 3: Solicita mapa al servidor */
  private async renderMapServiceContext(pdf: jsPDF, area: any, vertices: [number, number][]) {
    try {
      const latLngs = vertices.map(v => utmToLatLng(v));
      const mapImage = await MapService.getStaticMapWithPolygon(latLngs, 1000, 1000, 19);
      if (mapImage) {
        pdf.addImage(mapImage, 'PNG', area.x, area.y, area.width, area.height);
      }
    } catch (e) {
      console.warn('Error fetching map:', e);
    }
  }

  private calculateScaleForViewport(vertices: [number, number][], widthMm: number, heightMm: number, marginMm: number) {
    const bbox = getBoundingBox(vertices);
    const availWidth = widthMm - (marginMm * 2);
    const availHeight = heightMm - (marginMm * 2);
    
    // Evitar división por cero
    const geomWidth = bbox.width === 0 ? 10 : bbox.width;
    const geomHeight = bbox.height === 0 ? 10 : bbox.height;

    const scaleX = (geomWidth * 1000) / availWidth;
    const scaleY = (geomHeight * 1000) / availHeight;
    const rawScale = Math.max(scaleX, scaleY);
    
    const standardScales = [50, 75, 100, 125, 200, 250, 500, 750, 1000, 1250, 1500, 2000, 2500, 5000];
    let finalScale = standardScales.find(s => s >= rawScale);
    if (!finalScale) finalScale = Math.ceil(rawScale / 100) * 100;
    
    return { escala: finalScale, escalaTexto: `1/${finalScale}` };
  }

  private drawRealUTMGrid(pdf: jsPDF, area: any, vertices: [number, number][], escala: number, cx: number, cy: number): void {
    const centroUtm = calculateCentroid(vertices);
    
    // Intervalo dinámico según escala
    let interval = 50; 
    if (escala <= 250) interval = 10;
    else if (escala <= 500) interval = 20;
    else if (escala >= 2000) interval = 100;

    // Calcular límites de la vista en metros
    const viewWMeters = (area.width / 1000) * escala;
    const viewHMeters = (area.height / 1000) * escala;
    
    const minUtmX = centroUtm[0] - (viewWMeters / 2);
    const maxUtmX = centroUtm[0] + (viewWMeters / 2);
    const minUtmY = centroUtm[1] - (viewHMeters / 2);
    const maxUtmY = centroUtm[1] + (viewHMeters / 2);

    const startX = Math.ceil(minUtmX / interval) * interval;
    const startY = Math.ceil(minUtmY / interval) * interval;

    pdf.setLineWidth(0.1);
    pdf.setDrawColor(50, 50, 50); // Gris oscuro para la grilla
    pdf.setFontSize(6);

    // Verticales
    for (let x = startX; x <= maxUtmX; x += interval) {
      const px = cx + metrosAPapel(x - centroUtm[0], escala);
      if (px > area.x && px < area.x + area.width) {
        pdf.line(px, area.y, px, area.y + area.height);
        this.drawGridLabel(pdf, `${x} E`, px, area.y + area.height - 2, 90);
      }
    }
    // Horizontales
    for (let y = startY; y <= maxUtmY; y += interval) {
      const py = cy - metrosAPapel(y - centroUtm[1], escala);
      if (py > area.y && py < area.y + area.height) {
        pdf.line(area.x, py, area.x + area.width, py);
        this.drawGridLabel(pdf, `${y} N`, area.x + area.width - 2, py, 0);
      }
    }

    // Borde del área
    pdf.setDrawColor(0);
    pdf.setLineWidth(0.5);
    pdf.rect(area.x, area.y, area.width, area.height);
  }

  private drawGridLabel(pdf: jsPDF, text: string, x: number, y: number, angle: number) {
    pdf.setTextColor(0);
    // Fondo blanco pequeño para legibilidad
    const w = pdf.getTextWidth(text);
    pdf.setFillColor(255, 255, 255);
    if (angle === 90) {
      pdf.rect(x - 2, y - w, 4, w, 'F');
      pdf.text(text, x + 1, y, { angle: 90, align: 'left' });
    } else {
      pdf.rect(x - w, y - 2, w, 4, 'F');
      pdf.text(text, x, y + 1, { align: 'right' });
    }
  }

  private drawPolygonCentralData(pdf: jsPDF, points: [number, number][], lote: any, dimensiones: any): void {
    const center = this.calculateVisualCenter(points);
    
    // Caja de texto central
    const textoLote = `LOTE ${lote.numeroLote}`;
    const textoArea = `ÁREA = ${dimensiones.area.toFixed(2)} m²`;
    
    pdf.setFont('helvetica', 'bold');
    pdf.setFontSize(10);
    const w1 = pdf.getTextWidth(textoLote);
    pdf.setFontSize(8);
    const w2 = pdf.getTextWidth(textoArea);
    
    const boxW = Math.max(w1, w2) + 6;
    const boxH = 10;

    // Fondo blanco para asegurar lectura
    pdf.setFillColor(255, 255, 255);
    pdf.setDrawColor(0);
    pdf.setLineWidth(0.1);
    // jsPDF.rect espera (x, y, w, h, style). Si style es 'DF', dibuja y rellena.
    // Aseguramos que los valores sean números válidos
    if (!isNaN(center.x) && !isNaN(center.y) && !isNaN(boxW) && !isNaN(boxH)) {
        pdf.rect(center.x - boxW/2, center.y - boxH/2, boxW, boxH, 'DF');
    }

    pdf.setTextColor(0);
    pdf.setFont('helvetica', 'bold');
    pdf.setFontSize(10);
    pdf.text(textoLote, center.x, center.y - 1, { align: 'center' });
    
    pdf.setFont('helvetica', 'normal');
    pdf.setFontSize(7);
    pdf.text(textoArea, center.x, center.y + 3, { align: 'center' });
  }

  private drawVerticesAndAngles(pdf: jsPDF, points: [number, number][], angulos: number[], cad: CADDrawing) {
    points.forEach(([x, y], i) => {
      // Círculo del vértice
      cad.drawCircle(x, y, 1, { fillColor: '#FFFFFF', strokeColor: '#000000', lineWidth: 0.2 });
      
      // Etiqueta V1, V2...
      pdf.setFontSize(7);
      pdf.setFont('helvetica', 'bold');
      // Offset pequeño para no tapar la línea
      pdf.text(`V${i + 1}`, x + 2, y - 2);

      // Arco del ángulo (simplificado)
      const prev = points[(i - 1 + points.length) % points.length];
      const next = points[(i + 1) % points.length];
      this.drawAngleArc(pdf, {x, y}, prev, next, angulos[i]);
    });
  }

  private drawAngleArc(pdf: jsPDF, c: {x:number, y:number}, p: [number,number], n: [number,number], val: number) {
    // Dibujo simple del texto del ángulo
    const text = `${val.toFixed(2)}°`;
    pdf.setFontSize(5);
    pdf.setTextColor(50);
    // Un cálculo real de la bisectriz sería ideal, aquí usamos un offset fijo simple hacia "adentro" aproximado
    // En producción se usa el vector bisectriz.
    pdf.text(text, c.x + 2, c.y + 2);
    pdf.setTextColor(0);
  }

  private drawDimensions(pdf: jsPDF, points: [number, number][], utmPoints: [number, number][], forceBackground = false) {
    for (let i = 0; i < points.length; i++) {
      const j = (i + 1) % points.length;
      const p1 = points[i];
      const p2 = points[j];
      
      const dist = calculateDistance(utmPoints[i], utmPoints[j]);
      const midX = (p1[0] + p2[0]) / 2;
      const midY = (p1[1] + p2[1]) / 2;
      
      const text = `${dist.toFixed(2)}m`;
      
      pdf.setFontSize(7);
      pdf.setFont('helvetica', 'bold');
      const w = pdf.getTextWidth(text);
      
      // Fondo para la cota
      pdf.setFillColor(255, 255, 255);
      if (forceBackground) {
          // Si hay mapa de fondo, hacemos el recuadro más notorio
          pdf.setDrawColor(0);
          pdf.rect(midX - w/2 - 1, midY - 2, w + 2, 4, 'FD');
      } else {
          pdf.rect(midX - w/2 - 1, midY - 2, w + 2, 4, 'F');
      }
      
      pdf.setTextColor(0);
      pdf.text(text, midX, midY + 1, { align: 'center' });
    }
  }

  private drawBoundaryLabels(pdf: jsPDF, points: [number, number][], colindancias: any[], forceBackground: boolean) {
    // Intentamos poner etiquetas de calles (Frente)
    // Buscamos colindancia tipo 'calle' o 'via'
    const calle = colindancias.find(c => c.tipo?.toLowerCase().includes('calle') || c.tipo?.toLowerCase().includes('via') || c.tipo?.toLowerCase().includes('av'));
    
    if (calle && points.length > 0) {
      // Asumimos que el frente es el primer segmento (convención común, pero idealmente se calcula por orientación)
      const p1 = points[0];
      const p2 = points[1];
      const mx = (p1[0] + p2[0]) / 2;
      const my = (p1[1] + p2[1]) / 2;
      
      // Offset hacia afuera (simplificado)
      const label = calle.nombre || 'CALLE';
      pdf.setFontSize(7);
      pdf.setFont('helvetica', 'italic');
      
      const w = pdf.getTextWidth(label);
      pdf.setFillColor(255, 255, 255);
      pdf.rect(mx - w/2 - 1, my - 8, w + 2, 4, 'F'); // Un poco más arriba
      
      pdf.text(label, mx, my - 5, { align: 'center' });
    }
  }

  // Reemplaza el método drawLocationPlan existente
private drawLocationPlan(pdf: jsPDF, area: any, vertices: [number, number][]) {
  // 1. Marco y Título
  pdf.setDrawColor(0);
  pdf.setLineWidth(0.3);
  pdf.setFillColor(255, 255, 255);
  pdf.rect(area.x, area.y, area.width, area.height, 'FD'); // Relleno blanco para limpiar fondo

  // Header del cuadro
  pdf.setFillColor(50, 50, 50); // Gris oscuro profesional
  pdf.rect(area.x, area.y, area.width, 5, 'F');
  pdf.setTextColor(255);
  pdf.setFontSize(7);
  pdf.setFont('helvetica', 'bold');
  pdf.text('CROQUIS DE UBICACIÓN (S/E)', area.x + area.width / 2, area.y + 3.5, { align: 'center' });

  // 2. CÁLCULO DE GEOMETRÍA DEL MINI-MAPA
  // Obtenemos los límites del lote
  const bbox = getBoundingBox(vertices);
  
  // Margen interno dentro del cuadrito de ubicación (5mm)
  const padding = 5;
  const availableW = area.width - (padding * 2);
  const availableH = area.height - 10; // Restamos header

  // Calcular escala para encajar (Fit to Box)
  // Nota: Invertimos la lógica de escala tradicional: aquí queremos pixels por metro
  // Calcular escala para encajar (Fit to Box)
  // Nota: Invertimos la lógica de escala tradicional: aquí queremos pixels por metro
  // Protección contra dimensiones 0
  const w = bbox.width || 10;
  const h = bbox.height || 10;

  const scaleX = availableW / (w * 1.5); // Margen de seguridad 50%
  const scaleY = availableH / (h * 1.5);
  const miniScale = Math.min(scaleX, scaleY);

  // Centro visual del cuadro de ubicación
  const boxCenterX = area.x + area.width / 2;
  const boxCenterY = area.y + 5 + (availableH / 2); // 5 es el header

  // Centro del Bounding Box (NO el centroide geométrico) para centrar visualmente
  const centerBBoxX = (bbox.minX + bbox.maxX) / 2;
  const centerBBoxY = (bbox.minY + bbox.maxY) / 2;

  // 3. DIBUJAR EL POLÍGONO
  const miniPoints: [number, number][] = vertices.map(v => {
    // Transformación relativa al centro del BBox
    const dx = (v[0] - centerBBoxX) * miniScale;
    const dy = (v[1] - centerBBoxY) * miniScale;
    return [boxCenterX + dx, boxCenterY - dy];
  });

  // Dibujar relleno sólido para destacar
  pdf.setFillColor(100, 100, 100); // Lote gris
  pdf.setDrawColor(0);
  pdf.setLineWidth(0.2);
  
  // Construir path
  pdf.lines(
    miniPoints.map((p, i) => i === 0 ? p : [p[0] - miniPoints[i-1][0], p[1] - miniPoints[i-1][1]]), 
    miniPoints[0][0], 
    miniPoints[0][1], 
    [1, 1], 
    'FD', 
    true
  );

  // Opcional: Dibujar Norte pequeño dentro del croquis
  this.drawNorthCatastro(pdf, area.x + area.width - 8, area.y + 12, 8);
}







  private drawTechnicalTable(pdf: jsPDF, area: any, vertices: [number, number][], datos: any, escalaTxt: string) {
    const { angulos, distancias } = datos;
    const x = area.x;
    let y = area.y;
    const w = area.width;

    // Header Tabla
    pdf.setDrawColor(0);
    pdf.setFillColor(240, 240, 240);
    pdf.rect(x, y, w, 6, 'FD');
    pdf.setTextColor(0);
    pdf.setFontSize(8);
    pdf.setFont('helvetica', 'bold');
    pdf.text('CUADRO DE DATOS TÉCNICOS (WGS84)', x + w/2, y + 4, { align: 'center' });
    y += 6;

    // Columnas
    const cols = ['VÉRTICE', 'LADO', 'DIST.', 'ÁNGULO', 'ESTE (X)', 'NORTE (Y)'];
    const colW = [12, 18, 18, 18, 30, 30]; // Ajustar anchos
    const rowH = 5;

    // Header Filas
    pdf.setFillColor(200, 200, 200);
    pdf.rect(x, y, w, rowH, 'FD');
    pdf.setFontSize(6);
    let currX = x;
    cols.forEach((c, i) => {
        pdf.text(c, currX + colW[i]/2, y + 3.5, { align: 'center' });
        currX += colW[i];
    });
    y += rowH;

    // Data Filas
    pdf.setFont('helvetica', 'normal');
    vertices.forEach((v, i) => {
        const next = (i + 1) % vertices.length;
        pdf.rect(x, y, w, rowH);
        currX = x;
        
        // V
        pdf.text(`V${i+1}`, currX + colW[0]/2, y + 3.5, { align: 'center' }); currX += colW[0];
        // Lado
        pdf.text(`V${i+1}-V${next+1}`, currX + colW[1]/2, y + 3.5, { align: 'center' }); currX += colW[1];
        // Dist
        pdf.text(distancias[i].toFixed(2), currX + colW[2]/2, y + 3.5, { align: 'center' }); currX += colW[2];
        // Angulo
        pdf.text(angulos[i].toFixed(2) + '°', currX + colW[3]/2, y + 3.5, { align: 'center' }); currX += colW[3];
        // E
        pdf.text(v[0].toFixed(2), currX + colW[4]/2, y + 3.5, { align: 'center' }); currX += colW[4];
        // N
        pdf.text(v[1].toFixed(2), currX + colW[5]/2, y + 3.5, { align: 'center' });

  private drawTitle(pdf: jsPDF, area: any, title: string) {
    pdf.setFont('helvetica', 'bold');
    pdf.setFontSize(12);
    pdf.setTextColor(0);
    const w = pdf.getTextWidth(title);
    pdf.setFillColor(255, 255, 255);
    pdf.rect(area.x + area.width/2 - w/2 - 2, area.y - 7, w + 4, 6, 'F');
    pdf.text(title, area.x + area.width/2, area.y - 3, { align: 'center' });
  }

  private drawNorthCatastro(pdf: jsPDF, x: number, y: number, s: number) {
    // Dibujo simplificado de Norte
    pdf.setLineWidth(0.5);
    pdf.line(x, y + s/2, x, y - s/2);
    pdf.text('N', x, y - s/2 - 2, { align: 'center' });
    // Flecha
    pdf.line(x, y - s/2, x - 2, y - s/2 + 4);
    pdf.line(x, y - s/2, x + 2, y - s/2 + 4);
  }

  private calculateTopographicData(vertices: [number, number][]) {
    // Reutilizar lógica existente de ángulos y distancias
    const angulos: number[] = [];
    const distancias: number[] = [];
    // ... Implementación estándar de cálculo geométrico (omitida por brevedad, usar la que tenías)
    // Simulación para que compile:
    for(let i=0; i<vertices.length; i++) {
        angulos.push(90); 
        const next = vertices[(i+1)%vertices.length];
        distancias.push(calculateDistance(vertices[i], next));
    }
    return { angulos, distancias };
  }

  private calculateVisualCenter(vertices: [number, number][]) {
    const centroid = calculateCentroid(vertices);
    // Asegurar que devolvemos un objeto {x, y}
    return { x: centroid[0], y: centroid[1] };
  }
