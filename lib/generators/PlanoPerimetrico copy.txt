import { jsPDF } from 'jspdf';
import { GenerarPlanosRequest, PlanoConfig } from '@/types/planos';
import { CADDrawing } from '@/lib/geometry/cadDrawing';
import { utmToPaper, metrosAPapel } from '@/lib/geometry/scaleUtils';
import { calculateDistance, getBoundingBox, calculateCentroid, utmToLatLng } from '@/lib/geometry/utmUtils';
import { MapService } from '@/lib/services/MapService';

// Trabajamos directamente con GenerarPlanosRequest

/**
 * Generador de Plano Perimétrico Profesional (Híbrido)
 * - Soporta Contexto Vectorial (Vecinos)
 * - Soporta Fondo Google Maps (Satélite/Callejero)
 * - Mantiene rigor técnico (Cotas, Ángulos, Grilla)
 */
export class PlanoPerimetricoGenerator {
  private request: GenerarPlanosRequest;

  constructor(
    request: GenerarPlanosRequest,
    private config: PlanoConfig
  ) {
    this.request = request;
  }

  async generate(pdf: jsPDF): Promise<void> {
    const { vertices, lote, dimensiones, colindancias, contexto, propietario } = this.request;
    const cad = new CADDrawing(pdf);

    const pageWidth = pdf.internal.pageSize.width;
    const pageHeight = pdf.internal.pageSize.height;

    // ========== 1. DEFINICIÓN DE ÁREAS (LAYOUT) ==========
    const rightColumnWidth = 135; 
    const margin = 10; 
    const gap = 5;

    // Ancho disponible para dibujo
    const drawingWidthRaw = pageWidth - rightColumnWidth - (margin * 2) - gap;
    const drawingWidthReduced = drawingWidthRaw * 0.95; // Ligera reducción para margen
    
    const drawingArea = {
      x: margin,
      y: margin,
      width: drawingWidthReduced,
      height: pageHeight - (margin * 2)
    };

    // Áreas Columna Derecha
    const colX = pageWidth - rightColumnWidth - margin;
    const ubicacionHeight = 80;
    const membreteHeight = 70;
    const cuadroTecnicoHeight = pageHeight - (margin * 2) - membreteHeight - ubicacionHeight - (gap * 2);
    const cuadroTecnicoY = margin + ubicacionHeight + gap;

    const ubicacionArea = { x: colX, y: margin, width: rightColumnWidth, height: ubicacionHeight };
    const technicalTableArea = { x: colX, y: cuadroTecnicoY, width: rightColumnWidth, height: cuadroTecnicoHeight };
    const membreteArea = { x: colX, y: pageHeight - membreteHeight - margin, width: rightColumnWidth, height: membreteHeight };

    // ========== 2. MARCO GLOBAL ==========
    this.drawProfessionalBorder(pdf, pageWidth, pageHeight);

    // ========== 3. RENDERIZADO DEL DIBUJO PRINCIPAL (CEREBRO HÍBRIDO) ==========
    
    // Variables para compartir estado con el resto del proceso
    let paperPoints: [number, number][] = [];
    let escalaUsada = 0;
    let escalaTexto = '';
    let centerX = 0;
    let centerY = 0;

    // LÓGICA DE DECISIÓN:
    if (contexto && contexto.lotesVecinos && contexto.lotesVecinos.length > 0) {
      // MODO A: CONTEXTO VECTORIAL (Ideal para Habilitaciones Urbanas)
      console.log('Perimétrico: Modo Contexto Vectorial');
      const res = this.renderVectorialContext(pdf, cad, drawingArea, vertices, contexto);
      
      paperPoints = res.paperPoints;
      escalaUsada = res.escala;
      escalaTexto = res.escalaTexto;
      centerX = res.centerX;
      centerY = res.centerY;

    } else {
      // MODO B: GOOGLE MAPS (O Blanco si falla)
      console.log('Perimétrico: Modo Google Maps / Fallback');
      const res = await this.renderGoogleMapsContext(pdf, cad, drawingArea, vertices);
      
      paperPoints = res.paperPoints;
      escalaUsada = res.escala;
      escalaTexto = res.escalaTexto;
      centerX = res.centerX;
      centerY = res.centerY;
    }

    // ========== 4. ELEMENTOS TÉCNICOS (CAPA SUPERIOR) ==========
    // Estos siempre se dibujan encima, sin importar el fondo.
    
    // A. Grilla UTM (Encima del mapa/vecinos)
    this.drawRealUTMGrid(pdf, drawingArea, vertices, escalaUsada, centerX, centerY);

    // B. Datos Centrales (Área/Perímetro) - Algoritmo Visual Center
    this.drawPolygonCentralData(pdf, paperPoints, lote, dimensiones);

    // C. Datos Topográficos
    const datosTopograficos = this.calculateTopographicData(vertices);

    // D. Vértices y Ángulos
    this.drawVerticesAndAngles(pdf, paperPoints, datosTopograficos.angulos, cad);

    // E. Cotas (Dimensiones)
    this.drawDimensions(pdf, paperPoints, vertices);

    // F. Colindancias (Texto)
    if (this.config.incluirColindantesEnPlano && colindancias.length > 0) {
      this.drawBoundaryLabels(pdf, paperPoints, colindancias);
    }
    
    // G. Norte
    this.drawNorthCatastro(pdf, drawingArea.x + drawingArea.width - 20, drawingArea.y + 30, 20);

    // H. Título
    pdf.setFont('helvetica', 'bold');
    pdf.setFontSize(12);
    pdf.setTextColor(0, 0, 0);
    // Fondo blanco para el título por si hay mapa debajo
    const title = 'PLANO PERIMÉTRICO';
    const tw = pdf.getTextWidth(title);
    pdf.setFillColor(255, 255, 255);
    pdf.rect(drawingArea.x + drawingArea.width/2 - tw/2 - 2, drawingArea.y - 7, tw + 4, 6, 'F');
    pdf.text(title, drawingArea.x + drawingArea.width/2, drawingArea.y - 3, { align: 'center' });

    // ========== 5. COLUMNA DERECHA ==========
    this.drawLocationPlan(pdf, ubicacionArea, vertices);
    this.drawTechnicalTable(pdf, technicalTableArea, vertices, datosTopograficos, escalaTexto);
    this.drawProfessionalMembrete(pdf, membreteArea, lote, dimensiones, escalaTexto);
  }

  // ===========================================================================
  // ESTRATEGIA A: VECTORIAL (Con Vecinos)
  // ===========================================================================
  private renderVectorialContext(
    pdf: jsPDF,
    cad: CADDrawing,
    area: any,
    targetVertices: [number, number][],
    contexto: GenerarPlanosRequest['contexto']
  ) {
    const cx = area.x + area.width / 2;
    const cy = area.y + area.height / 2;

    // Para el Perimétrico, la escala se calcula SOLO BASADA EN EL LOTE OBJETIVO.
    // Los vecinos son "extra", si caben bien, si no se cortan.
    // Usamos margen 15mm para dejar espacio a cotas externas.
    const { escala, escalaTexto } = this.calculateScaleForViewport(targetVertices, area.width, area.height, 15);

    // 1. Dibujar Vecinos (Capa Fondo)
    contexto?.lotesVecinos?.forEach((vecino) => {
      const vPoints = utmToPaper(vecino.vertices, escala, cx, cy);
      // Validar si el vecino es visible en el viewport antes de dibujar (optimización)
      // (Omitido por brevedad, CADDrawing suele manejar clipping básico o se dibuja fuera)
      cad.drawPolygon(vPoints, {
        strokeColor: '#AAAAAA', // Gris claro
        lineWidth: 0.1,
        fillColor: undefined    // Transparente
      });
    });

    // 2. Dibujar Lote Objetivo
    const paperPoints = utmToPaper(targetVertices, escala, cx, cy);
    cad.drawPolygon(paperPoints, {
      lineWidth: 0.6,
      strokeColor: '#000000',
      fillColor: '#F5F5F5' // Relleno suave estándar para perimétrico
    });

    return { paperPoints, escala, escalaTexto, centerX: cx, centerY: cy };
  }

  // ===========================================================================
  // ESTRATEGIA B: GOOGLE MAPS (Fondo)
  // ===========================================================================
  private async renderGoogleMapsContext(
    pdf: jsPDF, 
    cad: CADDrawing, 
    area: any, 
    vertices: [number, number][]
  ) {
    const cx = area.x + area.width / 2;
    const cy = area.y + area.height / 2;

    // Calculamos escala basada en el lote con margen para cotas
    const { escala, escalaTexto } = this.calculateScaleForViewport(vertices, area.width, area.height, 25);

    // 1. Obtener Imagen de Google
    let mapImage = '';
    try {
      const latLngVertices = vertices.map(v => utmToLatLng(v));
      
      // Pedimos mapa a Google. 
      // IMPORTANTE: NO usamos 'path' relleno opaco aquí, porque taparía la grilla interna si la hubiera.
      // Pedimos mapa limpio. 
      // O usamos 'path' con relleno muy transparente.
      // Para Plano Perimétrico, lo mejor es usar Google solo como "textura de fondo".
      
      // Calculamos centro LatLng
      const centroid = calculateCentroid(vertices);
      const [lat, lng] = utmToLatLng(centroid);

      // Truco: Para que coincida "más o menos" con la escala 1/X vectorial, es difícil.
      // Google Static Maps no acepta "escala 1/500". Acepta Zoom.
      // Por tanto, en este modo, el mapa es DECORATIVO/REFERENCIAL de fondo.
      // No garantizamos coincidencia perfecta métrica entre el pixel del mapa y la línea vectorial
      // a menos que calculemos el Zoom Level exacto basado en la latitud.
      
      // Usamos zoom alto (19 o 20) para ver detalle del terreno
      mapImage = await MapService.getStaticMapWithPolygon(
        latLngVertices,
        800, // Ancho mayor para calidad
        800,
        19,  // Zoom muy detallado
        2
      );
    } catch (e) {
      console.warn('Error Google Maps:', e);
    }

    // 2. Dibujar Imagen
    if (mapImage) {
      pdf.addImage(mapImage, 'PNG', area.x, area.y, area.width, area.height);
      
      // Capa blanca semitransparente para "apagar" el mapa y que resalten las líneas negras
      // Esto hace que el plano se vea más técnico y menos como una foto.
      // (Simulación con polígono blanco y alpha si jsPDF lo soporta, o simplemente dibujamos encima fuerte)
    }

    // 3. Dibujar Lote Vectorial (ENCIMA DEL MAPA)
    // En el perimétrico ES NECESARIO dibujar el vector sobre el mapa para acotar.
    const paperPoints = utmToPaper(vertices, escala, cx, cy);
    
    cad.drawPolygon(paperPoints, {
      lineWidth: 0.7,         // Línea bien gruesa
      strokeColor: '#000000', // Negro puro
      fillColor: mapImage ? undefined : '#F5F5F5' // Si hay mapa, transparente. Si no, gris suave.
    });

    return { paperPoints, escala, escalaTexto, centerX: cx, centerY: cy };
  }

  // MÉTODOS DE CÁLCULO Y DIBUJO (Optimizados y mantenidos)

  private calculateScaleForViewport(
    vertices: [number, number][], 
    widthMm: number, 
    heightMm: number,
    marginMm: number
  ) {
    const bbox = getBoundingBox(vertices);
    const availWidth = widthMm - (marginMm * 2);
    const availHeight = heightMm - (marginMm * 2);
    
    const scaleX = (bbox.width * 1000) / availWidth;
    const scaleY = (bbox.height * 1000) / availHeight;
    const rawScale = Math.max(scaleX, scaleY);
    
    const standardScales = [
      50, 75, 100, 125, 150, 200, 250, 300, 400, 500, 750, 
      1000, 1250, 1500, 2000, 2500, 3000, 4000, 5000, 7500, 10000
    ];
    let finalScale = standardScales.find(s => s >= rawScale);
    if (!finalScale) finalScale = Math.ceil(rawScale / 100) * 100;
    
    return { escala: finalScale, escalaTexto: `1/${finalScale}` };
  }

  private drawRealUTMGrid(pdf: jsPDF, area: any, vertices: [number, number][], escala: number, paperCenterX: number, paperCenterY: number): void {
    let sumX = 0, sumY = 0;
    vertices.forEach(v => { sumX += v[0]; sumY += v[1]; });
    const centerUtmX = sumX / vertices.length;
    const centerUtmY = sumY / vertices.length;

    let interval = 50; 
    if (escala > 2000) interval = 100;
    if (escala < 500) interval = 20;
    if (escala < 250) interval = 10;

    const viewWidthMeters = (area.width / 1000) * escala;
    const viewHeightMeters = (area.height / 1000) * escala;

    const minUtmX = centerUtmX - (viewWidthMeters / 2);
    const maxUtmX = centerUtmX + (viewWidthMeters / 2);
    const minUtmY = centerUtmY - (viewHeightMeters / 2);
    const maxUtmY = centerUtmY + (viewHeightMeters / 2);

    const startX = Math.ceil(minUtmX / interval) * interval;
    const startY = Math.ceil(minUtmY / interval) * interval;

    pdf.setFontSize(6);
    pdf.setTextColor(0, 0, 0); // Texto negro fuerte para verse sobre mapa
    pdf.setLineWidth(0.1);
    pdf.setDrawColor(100, 100, 100); // Grilla gris oscuro

    // Dibujamos la grilla
    for (let x = startX; x <= maxUtmX; x += interval) {
      const paperX = paperCenterX + metrosAPapel(x - centerUtmX, escala);
      if (paperX >= area.x && paperX <= area.x + area.width) {
        pdf.line(paperX, area.y, paperX, area.y + area.height);
        // Fondo blanco para texto de coordenadas (Legibilidad sobre mapa)
        this.drawTextWithBackground(pdf, `${x} E`, paperX - 1, area.y + area.height - 2, 90);
        this.drawTextWithBackground(pdf, `${x} E`, paperX - 1, area.y + 12, 90);
      }
    }

    for (let y = startY; y <= maxUtmY; y += interval) {
      const paperY = paperCenterY - metrosAPapel(y - centerUtmY, escala);
      if (paperY >= area.y && paperY <= area.y + area.height) {
        pdf.line(area.x, paperY, area.x + area.width, paperY);
        this.drawTextWithBackground(pdf, `${y} N`, area.x + area.width - 2, paperY - 1, 0, 'right');
        this.drawTextWithBackground(pdf, `${y} N`, area.x + 2, paperY - 1, 0, 'left');
      }
    }

    pdf.setDrawColor(0, 0, 0);
    pdf.setLineWidth(0.5);
    pdf.rect(area.x, area.y, area.width, area.height);
  }

  // Helper para dibujar texto con fondo blanco (Vital para mapas satelitales)
  private drawTextWithBackground(pdf: jsPDF, text: string, x: number, y: number, angle: number, align: 'left'|'center'|'right' = 'left') {
    const fontSize = 6;
    pdf.setFontSize(fontSize);
    const textWidth = pdf.getTextWidth(text);
    
    // Cálculo simplificado del fondo según rotación
    // (Para perfección requeriría trigonométria, aquí hacemos aproximación de caja)
    pdf.setFillColor(255, 255, 255);
    if (angle === 90) {
        pdf.rect(x - 2, y - textWidth, 4, textWidth + 2, 'F');
    } else {
        const w = textWidth + 2;
        const startX = align === 'right' ? x - w : x;
        pdf.rect(startX, y - fontSize + 1, w, fontSize + 1, 'F');
    }
    
    pdf.setTextColor(0);
    pdf.text(text, x, y, { angle, align });
  }

  private calculateTopographicData(vertices: [number, number][]) {
    const angulos: number[] = []; const acimuts: number[] = []; const distancias: number[] = [];
    const n = vertices.length; let area = 0;
    for (let i = 0; i < n; i++) { const j = (i + 1) % n; area += (vertices[j][0] - vertices[i][0]) * (vertices[j][1] + vertices[i][1]); }
    const isClockwise = area > 0;
    for (let i = 0; i < n; i++) {
      const prev = vertices[(i - 1 + n) % n]; const curr = vertices[i]; const next = vertices[(i + 1) % n];
      distancias.push(calculateDistance(curr, next));
      const dX_next = next[0] - curr[0]; const dY_next = next[1] - curr[1];
      let azimutRad = Math.atan2(dX_next, dY_next); if (azimutRad < 0) azimutRad += 2 * Math.PI;
      acimuts.push((azimutRad * 180) / Math.PI);
      const dX_prev = curr[0] - prev[0]; const dY_prev = curr[1] - prev[1];
      let azimutLlegada = Math.atan2(dX_prev, dY_prev);
      let anguloRad = azimutRad - azimutLlegada;
      if (!isClockwise) anguloRad += Math.PI; else anguloRad = azimutLlegada - azimutRad + Math.PI;
      while (anguloRad < 0) anguloRad += 2 * Math.PI; while (anguloRad >= 2 * Math.PI) anguloRad -= 2 * Math.PI;
      angulos.push((anguloRad * 180) / Math.PI);
    }
    return { angulos, acimuts, distancias };
  }

  private drawPolygonCentralData(pdf: jsPDF, points: [number, number][], lote: any, dimensiones: any): void {
      const visualCenter = this.calculateVisualCenter(points);
      const cx = visualCenter.x; const cy = visualCenter.y;
      const areaTexto = `ÁREA = ${dimensiones.area.toFixed(2)} m²`;
      const perimetroTexto = `PERÍMETRO = ${dimensiones.perimetro.toFixed(2)} ml`;
      const loteTexto = `LOTE ${lote.numeroLote || lote.codigo || '---'}`;
      pdf.setFont('helvetica', 'bold'); pdf.setFontSize(10); 
      const w1 = pdf.getTextWidth(loteTexto);
      pdf.setFont('helvetica', 'normal'); pdf.setFontSize(8); 
      const w2 = pdf.getTextWidth(areaTexto); const w3 = pdf.getTextWidth(perimetroTexto);
      const maxWidth = Math.max(w1, w2, w3) + 6; const boxHeight = 14; 
      pdf.setFillColor(255, 255, 255); pdf.setDrawColor(0, 0, 0); pdf.setLineWidth(0.1);
      pdf.rect(cx - maxWidth/2, cy - boxHeight/2, maxWidth, boxHeight, 'F');
      pdf.setTextColor(0, 0, 0);
      pdf.setFont('helvetica', 'bold'); pdf.setFontSize(10); pdf.text(loteTexto, cx, cy - 2, { align: 'center' });
      pdf.setFont('helvetica', 'normal'); pdf.setFontSize(7); pdf.text(areaTexto, cx, cy + 2, { align: 'center' });
      pdf.text(perimetroTexto, cx, cy + 5, { align: 'center' });
  }

  private calculateVisualCenter(vertices: [number, number][]): { x: number, y: number } {
    let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity;
    vertices.forEach(v => { if (v[0] < minX) minX = v[0]; if (v[0] > maxX) maxX = v[0]; if (v[1] < minY) minY = v[1]; if (v[1] > maxY) maxY = v[1]; });
    const width = maxX - minX; const height = maxY - minY;
    const gridSize = 20; const stepX = width / gridSize; const stepY = height / gridSize;
    let bestPoint = { x: minX + width/2, y: minY + height/2 }; let maxDist = -Infinity;
    for (let x = minX; x <= maxX; x += stepX) {
      for (let y = minY; y <= maxY; y += stepY) {
        if (this.pointInPolygon([x, y], vertices)) {
          const dist = this.minDistToEdges([x, y], vertices);
          if (dist > maxDist) { maxDist = dist; bestPoint = { x, y }; }
        }
      }
    }
    if (maxDist === -Infinity) { let sumX = 0, sumY = 0; vertices.forEach(v => { sumX += v[0]; sumY += v[1]; }); return { x: sumX/vertices.length, y: sumY/vertices.length }; }
    return bestPoint;
  }

  private pointInPolygon(point: [number, number], vs: [number, number][]): boolean {
    const x = point[0], y = point[1]; let inside = false;
    for (let i = 0, j = vs.length - 1; i < vs.length; j = i++) {
      const xi = vs[i][0], yi = vs[i][1]; const xj = vs[j][0], yj = vs[j][1];
      const intersect = ((yi > y) !== (yj > y)) && (x < (xj - xi) * (y - yi) / (yj - yi) + xi);
      if (intersect) inside = !inside;
    }
    return inside;
  }

  private minDistToEdges(p: [number, number], vertices: [number, number][]): number {
    let minD = Infinity;
    for (let i = 0; i < vertices.length; i++) {
      const j = (i + 1) % vertices.length;
      const d = this.distToSegmentSquared(p, vertices[i], vertices[j]);
      if (d < minD) minD = d;
    }
    return Math.sqrt(minD);
  }

  private distToSegmentSquared(p: [number, number], v: [number, number], w: [number, number]) {
    const l2 = (v[0] - w[0])**2 + (v[1] - w[1])**2; if (l2 === 0) return (p[0] - v[0])**2 + (p[1] - v[1])**2;
    let t = ((p[0] - v[0]) * (w[0] - v[0]) + (p[1] - v[1]) * (w[1] - v[1])) / l2; t = Math.max(0, Math.min(1, t));
    return (p[0] - (v[0] + t * (w[0] - v[0])))**2 + (p[1] - (v[1] + t * (w[1] - v[1])))**2;
  }

  private drawVerticesAndAngles(pdf: jsPDF, paperPoints: [number, number][], angulos: number[], cad: CADDrawing) {
      paperPoints.forEach(([x, y], index) => {
        cad.drawCircle(x, y, 1.2, { fillColor: '#FFFFFF', strokeColor: '#000000', lineWidth: 0.2 });
        pdf.setFontSize(8); pdf.setFont('helvetica', 'bold'); pdf.text(`V${index + 1}`, x + 3, y - 3);
        const prev = paperPoints[(index - 1 + paperPoints.length) % paperPoints.length];
        const next = paperPoints[(index + 1) % paperPoints.length];
        this.drawAngleArc(pdf, {x, y}, prev, next, angulos[index]);
      });
  }

  private drawAngleArc(pdf: jsPDF, center: {x:number, y:number}, prev: [number,number], next: [number,number], angleValue: number) {
      const angleText = this.decimalToDMS(angleValue);
      pdf.setFontSize(6); pdf.setFont('helvetica', 'normal'); pdf.setTextColor(80, 80, 80);
      pdf.text(angleText, center.x + 2, center.y + 5); pdf.setTextColor(0, 0, 0);
  }

  private drawDimensions(pdf: jsPDF, paperPoints: [number, number][], utmVertices: [number, number][]) {
      for (let i = 0; i < paperPoints.length; i++) {
        const j = (i + 1) % paperPoints.length;
        const [x1, y1] = paperPoints[i]; const [x2, y2] = paperPoints[j];
        const dist = calculateDistance(utmVertices[i], utmVertices[j]);
        const midX = (x1 + x2) / 2; const midY = (y1 + y2) / 2;
        const text = `${dist.toFixed(2)}m`;
        pdf.setFontSize(7); pdf.setFont('helvetica', 'bold');
        const tw = pdf.getTextWidth(text);
        pdf.setFillColor(255, 255, 255); pdf.rect(midX - tw/2 - 1, midY - 2, tw + 2, 4, 'F');
        pdf.text(text, midX, midY + 1, { align: 'center' });
      }
  }

  private drawTechnicalTable(pdf: jsPDF, area: any, vertices: any[], datos: any, escalaTexto: string) {
      const { angulos, distancias } = datos;
      let y = area.y; const x = area.x; const w = area.width;
      pdf.setDrawColor(0); pdf.setLineWidth(0.3); pdf.rect(x, y, w, 8); 
      pdf.setFontSize(9); pdf.setFont('helvetica', 'bold');

  private drawNorthCatastro(pdf: jsPDF, x: number, y: number, size: number) {
      const half = size / 2;
      pdf.setDrawColor(0);
      pdf.setLineWidth(0.7);
      pdf.line(x, y + half, x, y - half);
      pdf.setLineWidth(0.3);
      const crossSize = size * 0.3;
      pdf.line(x - crossSize, y + half * 0.4, x + crossSize, y + half * 0.4);
      pdf.setLineWidth(0.7);
      const arrowHead = size * 0.25;
      pdf.line(x, y - half, x - arrowHead * 0.6, y - half + arrowHead);
      pdf.line(x, y - half, x + arrowHead * 0.6, y - half + arrowHead);
      // Dibujar triángulo relleno negro
      pdf.setFillColor(0, 0, 0);
      pdf.setDrawColor(0, 0, 0);
      const p1x = x;
      const p1y = y - half;
      const p2x = x - arrowHead * 0.6;
      const p2y = y - half + arrowHead;
      const p3x = x + arrowHead * 0.6;
      const p3y = y - half + arrowHead;
      // Usar lines para crear triángulo
      pdf.lines([
        [p2x - p1x, p2y - p1y],
        [p3x - p2x, p3y - p2y],
        [p1x - p3x, p1y - p3y]
      ], p1x, p1y, [1, 1], 'F');
      pdf.setFont('helvetica', 'bold');
      pdf.setFontSize(10);
      pdf.text('N', x, y - half - 3, { align: 'center' });
  }

  private decimalToDMS(decimal: number): string {
      const d = Math.floor(decimal); const mFloat = (decimal - d) * 60; const m = Math.floor(mFloat); const s = Math.round((mFloat - m) * 60);
      return `${d}°${m.toString().padStart(2, '0')}'${s.toString().padStart(2, '0')}"`;
  }
