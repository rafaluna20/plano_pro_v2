
import { jsPDF } from 'jspdf';
import { GenerarPlanosRequest, PlanoConfig } from '@/types/planos';
import { CADDrawing } from '@/lib/geometry/cadDrawing';
import { calculateCentroid, utmToLatLng, getBoundingBox } from '@/lib/geometry/utmUtils';
import { utmToPaper, metrosAPapel } from '@/lib/geometry/scaleUtils';
import { MapService } from '@/lib/services/MapService'; // Asegúrate de tener este servicio creado



// Trabajamos directamente con GenerarPlanosRequest

/**
 * Generador de Plano de Ubicación Profesional
 * - Prioridad 1: Contexto Vectorial (Habilitación Urbana) -> Costo 0, Calidad CAD.
 * - Prioridad 2: Google Maps (Lote Aislado) -> Usa Server-Side Rendering para alineación perfecta.
 */
export class PlanoUbicacionGenerator {
  private request: GenerarPlanosRequest;

  constructor(request: GenerarPlanosRequest, private config: PlanoConfig) {
    this.request = request;
  }

  async generate(pdf: jsPDF): Promise<void> {
    const { vertices, lote, dimensiones, propietario, contexto } = this.request;
    const cad = new CADDrawing(pdf);
    const pageWidth = pdf.internal.pageSize.width;
    const pageHeight = pdf.internal.pageSize.height;

    // ========== 1. CONFIGURACIÓN DE LAYOUT (A3 Landscape) ==========
    const rightColumnWidth = 135; 
    const margin = 10;
    const gap = 5;

    // Áreas de trabajo
    const drawingWidth = pageWidth - rightColumnWidth - (margin * 2) - gap;
    const drawingArea = {
      x: margin,
      y: margin,
      width: drawingWidth,
      height: pageHeight - (margin * 2)
    };

    const colX = pageWidth - rightColumnWidth - margin;
    // Alturas de paneles derechos
    const membreteHeight = 70;
    const infoHeight = 90;
    const coordsHeight = 60;
    // Espacio restante para macrolocalización
    const macroHeight = pageHeight - (margin * 2) - membreteHeight - infoHeight - coordsHeight - (gap * 3);

    const infoArea = { x: colX, y: margin, width: rightColumnWidth, height: infoHeight };
    const coordsArea = { x: colX, y: margin + infoHeight + gap, width: rightColumnWidth, height: coordsHeight };
    const macroArea = { x: colX, y: coordsArea.y + coordsHeight + gap, width: rightColumnWidth, height: macroHeight };
    const membreteArea = { x: colX, y: pageHeight - membreteHeight - margin, width: rightColumnWidth, height: membreteHeight };

    // ========== 2. DIBUJAR MARCO ==========
    this.drawProfessionalBorder(pdf, pageWidth, pageHeight);

    // ========== 3. RENDERIZADO INTELIGENTE (CEREBRO) ==========
    let escalaTexto = '';

    // LÓGICA DE DECISIÓN:
    if (contexto && contexto.lotesVecinos && contexto.lotesVecinos.length > 0) {
      // ESTRATEGIA A: VECTORIAL (Mejor Calidad / Costo Cero)
      console.log('Generando Ubicación: Modo Vectorial (Contexto Urbano)');
      const res = this.renderVectorialContext(pdf, cad, drawingArea, vertices, contexto);
      escalaTexto = res.escalaTexto;
    } else {
      // ESTRATEGIA B: GOOGLE MAPS (Mejorada con Overlay de Servidor)
      console.log('Generando Ubicación: Modo Google Maps (Lote Aislado)');
      const res = await this.renderGoogleMapsContext(pdf, cad, drawingArea, vertices);
      escalaTexto = res.escalaTexto;
    }

    // Título Central
    pdf.setFont('helvetica', 'bold');
    pdf.setFontSize(14);
    pdf.setTextColor(0, 0, 0);
    pdf.text('ESQUEMA DE LOCALIZACIÓN', drawingArea.x + drawingArea.width / 2, drawingArea.y + 10, { align: 'center' });

    // ========== 4. PANELES DE INFORMACIÓN ==========
    this.drawGeneralData(pdf, infoArea, lote, dimensiones);
    this.drawCentroidTable(pdf, coordsArea, vertices);
    
    // Macrolocalización (Placeholder o Mapa General estático del proyecto)
    this.drawMacrolocalization(pdf, macroArea);

    // Membrete
    this.drawProfessionalMembrete(pdf, membreteArea, lote, dimensiones, escalaTexto, propietario);
  }

  // ===========================================================================
  // ESTRATEGIA A: VECTORIAL (Habilitación Urbana)
  // ===========================================================================
  private renderVectorialContext(
    pdf: jsPDF,
    cad: CADDrawing,
    area: any,
    targetVertices: [number, number][],
    contexto: GenerarPlanosRequest['contexto']
  ) {
    // Calcular escala considerando vecinos para que se vea el entorno
    let allVertices = [...targetVertices];
    contexto?.lotesVecinos?.forEach(v => allVertices.push(...v.vertices));

    const cx = area.x + area.width / 2;
    const cy = area.y + area.height / 2;

    const { escala, escalaTexto } = this.calculateScaleForViewport(allVertices, area.width, area.height, 20);

    // 1. Dibujar Vecinos (Fondo sutil)
    contexto?.lotesVecinos?.forEach((vecino) => {
      const vPoints = utmToPaper(vecino.vertices, escala, cx, cy);
      cad.drawPolygon(vPoints, { strokeColor: '#999999', lineWidth: 0.1 });
    });

    // 2. Dibujar Lote Objetivo (Resaltado)
    const tPoints = utmToPaper(targetVertices, escala, cx, cy);
    cad.drawPolygon(tPoints, { 
      strokeColor: '#000000', 
      lineWidth: 0.5, 
      fillColor: '#808080' // Hatch sólido gris
    });

    // 3. Grilla y Norte
    this.drawGridContext(pdf, area, 20);
    cad.drawNorthSymbol(area.x + area.width - 25, area.y + 35, 20);

    return { escala, escalaTexto };
  }

  // ===========================================================================
  // ESTRATEGIA B: GOOGLE MAPS (Avanzado: Polygon Overlay)
  // ===========================================================================
  private async renderGoogleMapsContext(
    pdf: jsPDF, 
    cad: CADDrawing, 
    area: any, 
    vertices: [number, number][]
  ) {
    // Calcular escala referencial para mostrar en el membrete
    // Usamos margen amplio (40mm) para asegurar contexto visual
    const { escalaTexto } = this.calculateScaleForViewport(vertices, area.width, area.height, 40);

    let mapImage = '';
    
    try {
      // 1. Convertir UTM a LatLng (Necesario para Google)
      // Asumimos que la proyección es UTM 18S (Perú) o la configurada en tu utils
      const latLngVertices = vertices.map(v => utmToLatLng(v));

      // 2. Solicitar Mapa con Polígono Pintado por Google
      // Pedimos 640x640px (Max Free Tier) con escala @2x -> Calidad 1280px
      mapImage = await MapService.getStaticMapWithPolygon(
        latLngVertices,
        640,
        640
      );
    } catch (e) {
      console.warn('Error obteniendo mapa de Google:', e);
    }

    // 3. Renderizar
    if (mapImage) {
      // Dibujar imagen ajustada al área ("Cover" mode)
      // Como Google devuelve imagen cuadrada (normalmente), centramos o ajustamos
      pdf.addImage(mapImage, 'PNG', area.x, area.y, area.width, area.height);
      
      // Borde del área de dibujo
      pdf.setDrawColor(0);
      pdf.setLineWidth(0.4);
      pdf.rect(area.x, area.y, area.width, area.height);

      // NOTA: No dibujamos el polígono vectorial encima porque Google ya lo pintó.
      // Esto evita el "efecto fantasma" o desalineamiento.
      
    } else {
      // Fallback: Si no hay internet o API Key falla, dibujamos el lote vectorial simple
      this.drawGridContext(pdf, area, 20);
      pdf.text('Imagen satelital no disponible', area.x + area.width/2, area.y + area.height/2, {align:'center'});
      
      const cx = area.x + area.width / 2;
      const cy = area.y + area.height / 2;
      const { escala } = this.calculateScaleForViewport(vertices, area.width, area.height, 40);
      const paperPts = utmToPaper(vertices, escala, cx, cy);
      
      cad.drawPolygon(paperPts, { strokeColor: '#000000', lineWidth: 0.4, fillColor: '#808080' });
    }

    // Norte (Siempre necesario)
    cad.drawNorthSymbol(area.x + area.width - 25, area.y + 35, 20);

    return { escala: 0, escalaTexto }; // Escala es referencial en modo mapa
  }

  // ===========================================================================
  // MÉTODOS AUXILIARES
  // ===========================================================================

  private calculateScaleForViewport(vertices: [number, number][], widthMm: number, heightMm: number, marginMm: number) {
    const bbox = getBoundingBox(vertices);
    const availWidth = widthMm - (marginMm * 2);
    const availHeight = heightMm - (marginMm * 2);
    
    const scaleX = (bbox.width * 1000) / availWidth;
    const scaleY = (bbox.height * 1000) / availHeight;
    const rawScale = Math.max(scaleX, scaleY);
    
    const standardScales = [100, 200, 250, 500, 750, 1000, 1250, 1500, 2000, 2500, 5000, 7500, 10000];
    let finalScale = standardScales.find(s => s >= rawScale);
    if (!finalScale) finalScale = Math.ceil(rawScale / 100) * 100;
    
    return { escala: finalScale, escalaTexto: `1/${finalScale}` };
  }

  private drawGridContext(pdf: jsPDF, area: any, step: number) {
     pdf.setDrawColor(220); pdf.setLineWidth(0.1);
     for(let x = area.x; x <= area.x + area.width; x += step) pdf.line(x, area.y, x, area.y + area.height);
     for(let y = area.y; y <= area.y + area.height; y += step) pdf.line(area.x, y, area.x + area.width, y);
     pdf.setDrawColor(0); pdf.setLineWidth(0.4); pdf.rect(area.x, area.y, area.width, area.height);
  }

  private drawProfessionalBorder(pdf: jsPDF, w: number, h: number) {
    pdf.setLineWidth(1); pdf.rect(5, 5, w - 10, h - 10);
    pdf.setLineWidth(0.2); pdf.rect(8, 8, w - 16, h - 16);
  }

  private drawGeneralData(pdf: jsPDF, area: any, lote: any, dimensiones: any) {
    const { x, y, width } = area;
    let currY = y;
    
    pdf.setFillColor(0, 0, 0); pdf.rect(x, currY, width, 8, 'F');
    pdf.setTextColor(255); pdf.setFontSize(9); pdf.setFont('helvetica', 'bold');
    pdf.text('DATOS GENERALES', x + width / 2, currY + 5, { align: 'center' });
    currY += 12;

    pdf.setTextColor(0); pdf.setFontSize(8);
    const items = [
      { l: 'Departamento:', v: 'Lima' },
      { l: 'Provincia:', v: 'Lima' },
      { l: 'Distrito:', v: lote.etapa || '---' },
      { l: 'Manzana:', v: lote.manzana || '---' },
      { l: 'Lote:', v: lote.numeroLote || '---' },
      { l: 'Área:', v: `${dimensiones.area.toFixed(2)} m²` },
    ];
    
    items.forEach((item, i) => {
        if (i % 2 !== 0) { pdf.setFillColor(245, 245, 245); pdf.rect(x, currY - 4.5, width, 6, 'F'); }
        pdf.setFont('helvetica', 'bold'); pdf.text(item.l, x + 2, currY);
        pdf.setFont('helvetica', 'normal'); pdf.text(item.v, x + width/2, currY);
        currY += 6;
    });
  }

  private drawCentroidTable(pdf: jsPDF, area: any, vertices: [number, number][]) {
    const centroid = calculateCentroid(vertices);
    const [lat, lng] = utmToLatLng(centroid);
    const { x, y, width } = area;
    let currY = y;

    pdf.setFillColor(0, 0, 0); pdf.rect(x, currY, width, 8, 'F');
    pdf.setTextColor(255); pdf.setFontSize(9); pdf.setFont('helvetica', 'bold');
    pdf.text('COORDENADAS (CENTROIDE)', x + width / 2, currY + 5, { align: 'center' });
    currY += 12;
    
    pdf.setTextColor(0); pdf.setFontSize(8);
    pdf.setFont('helvetica', 'bold'); pdf.text('UTM WGS84:', x + 2, currY);
    pdf.setFont('helvetica', 'normal'); pdf.text(`E: ${centroid[0].toFixed(2)} N: ${centroid[1].toFixed(2)}`, x + width/2 - 10, currY);
    currY += 6;
    pdf.setFont('helvetica', 'bold'); pdf.text('GEOGRÁFICAS:', x + 2, currY);
    pdf.setFont('helvetica', 'normal'); pdf.text(`Lat: ${lat.toFixed(5)} Lng: ${lng.toFixed(5)}`, x + width/2 - 10, currY);
  }

  private drawMacrolocalization(pdf: jsPDF, area: any) {
    const { x, y, width, height } = area;
    pdf.setDrawColor(0); pdf.setLineWidth(0.3); pdf.rect(x, y, width, height);
    pdf.setFillColor(0, 0, 0); pdf.rect(x, y, width, 6, 'F');
    pdf.setTextColor(255); pdf.setFontSize(8); pdf.text('MACRO LOCALIZACIÓN', x + width/2, y + 4, { align: 'center' });
    
    // Placeholder texto
    pdf.setTextColor(150); pdf.setFontSize(9); 
    pdf.text('VISTA GENERAL PROYECTO', x + width/2, y + height/2, {align:'center'});
  }

  private drawProfessionalMembrete(pdf: jsPDF, area: any, lote: any, dimensiones: any, escala: string, prop: any) {
    const { x, y, width: w, height: h } = area;
    pdf.setDrawColor(0); pdf.setLineWidth(0.5); pdf.rect(x, y, w, h);
    const leftW = w * 0.6; pdf.line(x + leftW, y, x + leftW, y + h);

    pdf.setTextColor(0); pdf.setFontSize(14); pdf.setFont('helvetica', 'bold');
    pdf.text('PLANO DE UBICACIÓN', x + leftW/2, y + 10, { align: 'center' });

    let lineY = y + 20; pdf.setLineWidth(0.2); pdf.line(x, lineY, x + leftW, lineY);
    pdf.setFontSize(7); pdf.text(`PROPIETARIO: ${prop?.nombre || '---'}`, x + 2, lineY + 5);
    
    lineY += 10; pdf.line(x, lineY, x + leftW, lineY);
    pdf.text(`UBICACIÓN: Mz. ${lote.manzana} Lt. ${lote.numeroLote}`, x + 2, lineY + 5);

    const footerY = y + h - 8; pdf.setLineWidth(0.5); pdf.line(x, footerY, x + w, footerY);
    pdf.text(`FECHA: ${new Date().toLocaleDateString()}`, x + 5, footerY + 5);
    pdf.text(`ESCALA: ${escala}`, x + w - 5, footerY + 5, { align: 'right' });
  }
}
