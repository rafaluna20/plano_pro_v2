'use client';

import { useState } from 'react';
import { useRouter } from 'next/navigation';
import { ProtectedRoute } from '@/components/auth/ProtectedRoute';
import { useEditorStore } from '@/lib/editor/store';
import { EditorMode } from '@/types/editor';
import { EditorWizard } from '@/components/editor/wizard/EditorWizard';
import { FreeEditor } from '@/components/editor/canvas/FreeEditor';
import toast, { Toaster } from 'react-hot-toast';

export default function NuevoEditorPage() {
  const router = useRouter();
  const { mode, setMode } = useEditorStore();
  const [showEditor, setShowEditor] = useState(false);

  const handleContinue = () => {
    if (!mode) return;

    setShowEditor(true);
    const modeLabel =
      mode === EditorMode.WIZARD ? 'Wizard guiado' :
        mode === EditorMode.FREEFORM ? 'Edición libre CAD' :
          mode === EditorMode.TEMPLATE ? 'Selector de plantillas' :
            'Importador de datos';
    toast.success(`Iniciando ${modeLabel}...`);
  };

  // Si ya seleccionó modo y continuó, mostrar el editor Wizard
  if (showEditor && mode === EditorMode.WIZARD) {
    return (
      <ProtectedRoute>
        <Toaster position="top-right" />
        <div className="h-screen flex flex-col">
          {/* Header */}
          <header className="bg-white border-b border-gray-200 shadow-sm">
            <div className="px-6 py-4 flex items-center justify-between">
              <div className="flex items-center gap-4">
                <button
                  onClick={() => setShowEditor(false)}
                  className="text-gray-600 hover:text-gray-900 transition-colors duration-200"
                  aria-label="Volver a selección"
                >
                  <svg
                    className="w-5 h-5"
                    fill="none"
                    strokeLinecap="round"
                    strokeLinejoin="round"
                    strokeWidth="2"
                    viewBox="0 0 24 24"
                    stroke="currentColor"
                  >
                    <path d="M15 19l-7-7 7-7"></path>
                  </svg>
                </button>
                <div>
                  <h1 className="text-xl font-bold text-gray-900">
                    Wizard Guiado - Nuevo Plano
                  </h1>
                  <p className="text-sm text-gray-600">
                    Proceso paso a paso para crear tu plano
                  </p>
                </div>
              </div>

              <button
                onClick={() => router.push('/dashboard')}
                className="px-4 py-2 text-sm font-medium text-gray-700 bg-white border border-gray-300 rounded-md hover:bg-gray-50 transition-all duration-200"
              >
                Salir
              </button>
            </div>
          </header>

          {/* Wizard Component */}
          <div className="flex-1 overflow-hidden">
            <EditorWizard />
          </div>
        </div>
      </ProtectedRoute>
    );
  }

  // Si ya seleccionó modo Edición Libre, mostrar el FreeEditor
  if (showEditor && mode === EditorMode.FREEFORM) {
    return (
      <ProtectedRoute>
        <Toaster position="top-right" />
        <FreeEditor />
      </ProtectedRoute>
    );
  }


  return (
    <ProtectedRoute>
      <Toaster position="top-right" />

      <div className="min-h-screen bg-gray-50">
        {/* Header */}
        <header className="bg-white border-b border-gray-200 shadow-sm sticky top-0 z-10">
          <div className="px-6 py-4 flex items-center justify-between">
            <div className="flex items-center gap-4">
              <button
                onClick={() => router.push('/dashboard')}
                className="text-gray-600 hover:text-gray-900 transition-colors duration-200"
                aria-label="Volver al dashboard"
              >
                <svg
                  className="w-5 h-5"
                  fill="none"
                  strokeLinecap="round"
                  strokeLinejoin="round"
                  strokeWidth="2"
                  viewBox="0 0 24 24"
                  stroke="currentColor"
                >
                  <path d="M15 19l-7-7 7-7"></path>
                </svg>
              </button>
              <div>
                <h1 className="text-xl font-bold text-gray-900">
                  Nuevo Plano - Editor Profesional
                </h1>
                <p className="text-sm text-gray-600">
                  Crea planos con herramientas profesionales CAD-like
                </p>
              </div>
            </div>

            <div className="flex gap-3">
              <button
                onClick={() => router.push('/dashboard')}
                className="px-4 py-2 text-sm font-medium text-gray-700 bg-white border border-gray-300 rounded-md hover:bg-gray-50 transition-all duration-200"
              >
                Cancelar
              </button>
              <button
                className="px-4 py-2 text-sm font-medium text-white bg-blue-600 hover:bg-blue-700 rounded-md transition-all duration-200 shadow-sm hover:shadow-md"
                onClick={() => toast.success('¡Función en desarrollo!')}
              >
                💾 Guardar Borrador
              </button>
            </div>
          </div>
        </header>

        {/* Main Content */}
        <div className="max-w-4xl mx-auto mt-12 px-6 pb-12">
          <div className="bg-white rounded-lg shadow-md p-8">
            <div className="mb-8">
              <h2 className="text-2xl font-bold text-gray-900 mb-2">
                Selecciona el Modo de Trabajo
              </h2>
              <p className="text-gray-600">
                Elige cómo deseas crear tu nuevo plano
              </p>
            </div>

            {/* Mode Selection Grid */}
            <div className="grid grid-cols-1 md:grid-cols-2 gap-6 mb-8">
              {/* Modo Wizard */}
              <button
                onClick={() => {
                  setMode(EditorMode.WIZARD);
                  toast.success('Modo Wizard activado');
                }}
                className={`p-6 border-2 rounded-lg transition-all duration-200 text-left ${mode === EditorMode.WIZARD
                  ? 'border-blue-600 bg-blue-50 shadow-md'
                  : 'border-gray-200 hover:border-blue-300 hover:shadow-sm'
                  }`}
              >
                <div className="text-4xl mb-3">🧙‍♂️</div>
                <h3 className="text-lg font-semibold text-gray-900 mb-2">
                  Wizard Guiado
                </h3>
                <p className="text-sm text-gray-600 mb-3">
                  Proceso paso a paso ideal para principiantes. Te guiaremos en cada etapa.
                </p>
                <div className="flex items-center gap-2 text-xs text-gray-500">
                  <span className="px-2 py-1 bg-green-100 text-green-700 rounded">
                    Recomendado
                  </span>
                  <span>5 pasos simples</span>
                </div>
                {mode === EditorMode.WIZARD && (
                  <div className="mt-4 flex items-center gap-2 text-sm font-medium text-blue-600">
                    <svg className="w-4 h-4" fill="currentColor" viewBox="0 0 20 20">
                      <path fillRule="evenodd" d="M10 18a8 8 0 100-16 8 8 0 000 16zm3.707-9.293a1 1 0 00-1.414-1.414L9 10.586 7.707 9.293a1 1 0 00-1.414 1.414l2 2a1 1 0 001.414 0l4-4z" clipRule="evenodd" />
                    </svg>
                    Seleccionado
                  </div>
                )}
              </button>

              {/* Modo Libre */}
              <button
                onClick={() => {
                  setMode(EditorMode.FREEFORM);
                  toast.success('Modo Libre activado');
                }}
                className={`p-6 border-2 rounded-lg transition-all duration-200 text-left ${mode === EditorMode.FREEFORM
                  ? 'border-blue-600 bg-blue-50 shadow-md'
                  : 'border-gray-200 hover:border-blue-300 hover:shadow-sm'
                  }`}
              >
                <div className="text-4xl mb-3">✏️</div>
                <h3 className="text-lg font-semibold text-gray-900 mb-2">
                  Edición Libre
                </h3>
                <p className="text-sm text-gray-600 mb-3">
                  Control total con herramientas profesionales. Para usuarios avanzados.
                </p>
                <div className="flex items-center gap-2 text-xs text-gray-500">
                  <span className="px-2 py-1 bg-purple-100 text-purple-700 rounded">
                    Avanzado
                  </span>
                  <span>Herramientas CAD</span>
                </div>
                {mode === EditorMode.FREEFORM && (
                  <div className="mt-4 flex items-center gap-2 text-sm font-medium text-blue-600">
                    <svg className="w-4 h-4" fill="currentColor" viewBox="0 0 20 20">
                      <path fillRule="evenodd" d="M10 18a8 8 0 100-16 8 8 0 000 16zm3.707-9.293a1 1 0 00-1.414-1.414L9 10.586 7.707 9.293a1 1 0 00-1.414 1.414l2 2a1 1 0 001.414 0l4-4z" clipRule="evenodd" />
                    </svg>
                    Seleccionado
                  </div>
                )}
              </button>

            </div>

            {/* Action Button */}
            <div className="flex justify-end">
              <button
                onClick={handleContinue}
                disabled={!mode}
                className={`px-6 py-3 text-base font-medium text-white rounded-md transition-all duration-200 ${mode
                  ? 'bg-blue-600 hover:bg-blue-700 shadow-md hover:shadow-lg'
                  : 'bg-gray-300 cursor-not-allowed'
                  }`}
              >
                Continuar →
              </button>
            </div>
          </div>

          {/* Info Card */}
          <div className="mt-6 bg-blue-50 border border-blue-200 rounded-lg p-4">
            <div className="flex gap-3">
              <div className="flex-shrink-0">
                <svg className="w-6 h-6 text-blue-600" fill="currentColor" viewBox="0 0 20 20">
                  <path fillRule="evenodd" d="M18 10a8 8 0 11-16 0 8 8 0 0116 0zm-7-4a1 1 0 11-2 0 1 1 0 012 0zM9 9a1 1 0 000 2v3a1 1 0 001 1h1a1 1 0 100-2v-3a1 1 0 00-1-1H9z" clipRule="evenodd" />
                </svg>
              </div>
              <div>
                <h4 className="text-sm font-semibold text-blue-900 mb-2">
                  💡 ¿Necesitas ayuda para decidir?
                </h4>
                <ul className="text-sm text-blue-800 space-y-1">
                  <li className="flex items-start gap-2">
                    <span className="text-blue-600 font-bold">•</span>
                    <span><strong>Wizard:</strong> Primera vez o proceso simple con guía paso a paso</span>
                  </li>
                  <li className="flex items-start gap-2">
                    <span className="text-blue-600 font-bold">•</span>
                    <span><strong>Libre:</strong> Usuario experimentado con herramientas CAD profesionales</span>
                  </li>
                </ul>
              </div>
            </div>
          </div>
        </div>
      </div>
    </ProtectedRoute>
  );
}
